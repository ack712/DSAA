#include <stdio.h>

#define MAX 5   // Maximum number of elements in the queue

int queue[MAX]; // Array used to store queue elements
int front = -1; // Points to the first element
int rear = -1;  // Points to the last element

// Function to insert an element into the queue
void enqueue()
{
    int value;

    // Check whether the queue is full
    if (rear == MAX - 1)
    {
        printf("Queue Overflow\n");
        return;
    }

    printf("Enter value to insert: ");
    scanf("%d", &value);

    // For the first element, set front to 0
    if (front == -1)
    {
        front = 0;
    }

    rear = rear + 1;   // Move rear to the next position
    queue[rear] = value; // Insert value at rear

    printf("%d inserted into the queue\n", value);
}

// Function to remove an element from the queue
void dequeue()
{
    // Check whether the queue is empty
    if (front == -1 || front > rear)
    {
        printf("Queue Underflow\n");
        return;
    }

    printf("%d removed from the queue\n", queue[front]);

    front = front + 1; // Move front to the next element

    // Reset queue when all elements are removed
    if (front > rear)
    {
        front = -1;
        rear = -1;
    }
}

// Function to display queue elements
void display()
{
    int i;

    if (front == -1)
    {
        printf("Queue is empty\n");
        return;
    }

    printf("Queue elements are:\n");

    // Display elements from front to rear
    for (i = front; i <= rear; i++)
    {
        printf("%d ", queue[i]);
    }

    printf("\n");
}

int main()
{
    int choice;

    do
    {
        printf("\n--- Queue Using Array ---\n");
        printf("1. Enqueue\n");
        printf("2. Dequeue\n");
        printf("3. Display\n");
        printf("4. Exit\n");

        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
            case 1:
                enqueue();
                break;

            case 2:
                dequeue();
                break;

            case 3:
                display();
                break;

            case 4:
                printf("Program ended\n");
                break;

            default:
                printf("Invalid choice\n");
        }

    } while (choice != 4);

    return 0;
}
