#include <stdio.h>
#include <stdlib.h>

// Structure of one node
struct Node
{
    int data;           // Stores the value
    struct Node *next;  // Address of next node
};

struct Node *head = NULL;

// Insert a new node at the end
void insert()
{
    int value;

    printf("Enter value: ");
    scanf("%d", &value);

    // Create a new node
    struct Node *newNode =
        (struct Node *)malloc(sizeof(struct Node));

    newNode->data = value;

    // If list is empty
    if (head == NULL)
    {
        head = newNode;

        // Node points back to itself
        newNode->next = head;
    }
    else
    {
        struct Node *temp = head;

        // Find the last node
        while (temp->next != head)
        {
            temp = temp->next;
        }

        temp->next = newNode;

        // Last node points back to head
        newNode->next = head;
    }

    printf("Node inserted.\n");
}

// Display all nodes
void display()
{
    struct Node *temp;

    if (head == NULL)
    {
        printf("List is empty.\n");
        return;
    }

    temp = head;

    printf("Circular Linked List: ");

    // Display until we return to head
    do
    {
        printf("%d -> ", temp->data);
        temp = temp->next;

    } while (temp != head);

    printf("HEAD\n");
}

int main()
{
    int choice;

    do
    {
        printf("\n1. Insert");
        printf("\n2. Display");
        printf("\n3. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
            case 1:
                insert();
                break;

            case 2:
                display();
                break;

            case 3:
                printf("Program ended.\n");
                break;

            default:
                printf("Invalid choice.\n");
        }

    } while (choice != 3);

    return 0;
}
